% To use your own font, replace the two blocks below with e.g.
%   \setmainfont{EB Garamond}   or   \setsansfont{Source Sans 3}

    BoldItalicFont = *-BoldItalic]
}{}

%==================================================
% Colors & styling
%   One palette for every document. Change a color here and the
%   letter, statements, samples and packet all follow.
%
%   What each color does:
%     color1  your name, section headings, document titles
%     color2  accent: the rule under your name, links, and the
%             highlighted (tailored) text in draft mode
%     color3  secondary text: contact line, dates, "Prepared for" line
%     color4  thin rules under section headings
%     color5  the DRAFT watermark (keep it pale)
%
%   How to change them:
%     \definecolor{color1}{HTML}{1B2A4A}       HEX code, no #
%     \definecolor{color1}{RGB}{27,42,74}      0-255 values
%     \definecolor{color1}{rgb}{0.11,0.16,0.29} 0-1 values
%   Good places to find palettes: coolors.co, colorhunt.co, or the
%   brand colors of your university or company. Check that color1
%   and color3 stay dark enough to read when printed in grayscale.
%==================================================

%==================================================
% Letterhead
%   \makeletterhead prints your name, an accent rule and one line
%   of contact details. Any detail left empty in applicant.tex is
%   skipped, along with its separator.
%==================================================
\newif\iffirstcontact
\newcommand{\contactsep}{\quad{\color{color4}|}\quad}
\newcommand{\contactitem}[1]{%
  \ifdefvoid{#1}{}{\iffirstcontact\firstcontactfalse\else\contactsep\fi#1}}
\newcommand{\contactlink}[2]{% #1 = field, #2 = url prefix
  \ifdefvoid{#1}{}{\iffirstcontact\firstcontactfalse\else\contactsep\fi
    \href{#2#1}{\color{color3}#1}}}

\newcommand{\makeletterhead}{%
  \begingroup
  \setstretch{1}\noindent
  {\sffamily\fontsize{24}{28}\selectfont\color{color1}%
    \applicantfirst\ \textbf{\applicantlast}%
    \ifdefvoid{\applicantcredentials}{}{%
      {\fontsize{14}{16}\selectfont\color{color3}\ \applicantcredentials}}}\par
  \vspace{4pt}\noindent{\color{color2}\rule{\linewidth}{1.4pt}}\par
  \vspace{2pt}\noindent
  {\sffamily\small\color{color3}\firstcontacttrue
    \contactitem{\applicantlocation}%
    \contactlink{\applicantemail}{mailto:}%
    \contactitem{\applicantphone}%
    \contactlink{\applicantwebsite}{https://}%
    \contactlink{\applicantlinkedin}{https://}%
    \contactlink{\applicantgithub}{https://}}\par
  \endgroup
  \vspace{14pt}}
